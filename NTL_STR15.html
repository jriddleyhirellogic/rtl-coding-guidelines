NTL_STR15

Message: Give unique names to synchronizers so that they can be identified

Description This rule fires when Leda detects a synchronizer in the design that does not have a unique label. Metastable circuits are handled seriously in timing analysis and synthesis. Naming synchronizers with unique label on their output pin makes them easier to identify. You can configure this rule using the rule_set_parameter Tcl command and the SYNCHRONIZER_NAME parameter to specify the unique name to be checked. The default value is "_sync". For example:
leda> rule_set_parameter -rule NTL_STR15 \
-parameter SYNCHRONIZER_NAME -value {_sn}

Policy DESIGN
Ruleset STRUCTURE
Language VHDL/Verilog
Type Hardware
Severity Error

Example

Here are valid and invalid Verilog code example, that illustrates the problem: 

module bad_str15 (a, b, o,Q2,D);
    input a, b;
    output o;
    wire a,b,o;
    wire i;
    output Q2;
    input  D;
    wire  inter2, inter3;

    and(i,a,b);
    or (o,i,b);

    FF_U U3 (inter2,D,i);
    FF_U U4(inter3,inter2,a);
    FF_U U5(Q2,inter3,a);

endmodule
 
module FF_U ( Q_a, D, CP);
  input D, CP;
  output Q_a;
  reg Q_a;

  always @ (posedge CP) begin
      Q_a = D; // NTL_STR15 fires -- Q_a is not having the unique name
end

endmodule


module good_str15 (a, b, o,Q2,D);
    input a, b;
    output o;
    wire a,b,o;
    wire i;
    output Q2;
    input  D;
    wire  inter2, inter3;

    and(i,a,b);
    or (o,i,b);

    FF_U U3 (inter2,D,i);
    FF_U U4(inter3,inter2,a);
    FF_U U5(Q2,inter3,a);

endmodule

module FF_U ( Q_sync, D, CP);
  input D, CP;
  output Q_sync;
  reg Q_sync;

  always @ (posedge CP) begin
     Q_sync = D;    // OK -- Q_sync is an unique name (default is _sync)
  end

endmodule
